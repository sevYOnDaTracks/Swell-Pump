cmake_minimum_required(VERSION 3.22)
project(SwellPump VERSION 0.1.0 LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(PUMP_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND PUMP_FORMATS AU)
endif()
juce_add_plugin(SwellPump COMPANY_NAME "Swell-audio" BUNDLE_ID "fr.swell-audio.pump" PLUGIN_MANUFACTURER_CODE Swel PLUGIN_CODE Spmp FORMATS ${PUMP_FORMATS} PRODUCT_NAME "Swell Pump" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_add_binary_data(PumpAssets SOURCES swell-mark.png)
target_sources(SwellPump PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(SwellPump PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(SwellPump PRIVATE PumpAssets juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
enable_testing()
juce_add_console_app(PumpTest PRODUCT_NAME "Pump Audio Tests")
target_sources(PumpTest PRIVATE Tests/AudioTest.cpp)
target_include_directories(PumpTest PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(PumpTest PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(PumpTest PRIVATE SwellPump)
add_test(NAME PumpAudio COMMAND PumpTest)
juce_add_console_app(PumpSnapshot PRODUCT_NAME "Pump Snapshot")
target_sources(PumpSnapshot PRIVATE Tools/Snapshot.cpp)
target_include_directories(PumpSnapshot PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(PumpSnapshot PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(PumpSnapshot PRIVATE SwellPump)
juce_add_console_app(PumpDemo PRODUCT_NAME "Pump Demo Renderer")
target_sources(PumpDemo PRIVATE Tools/RenderDemo.cpp)
target_include_directories(PumpDemo PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(PumpDemo PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(PumpDemo PRIVATE SwellPump)
